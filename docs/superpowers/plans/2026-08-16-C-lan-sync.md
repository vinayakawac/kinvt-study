# Sub-project C — Device-to-Device Sync Over Your Own Network

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Phone and desktop exchange progress directly over the local network, with no account, no server, and no cloud service — and an attacker on the same Wi-Fi learns nothing and can forge nothing.

**Architecture:** The desktop opens a short-lived HTTP listener and shows a QR code carrying its address and a fresh 256-bit key. The phone scans it, so **the key crosses optically and never touches the network**. That is what makes plain HTTP acceptable as transport: the payload itself is AES-256-GCM encrypted end to end. Both sides then merge with the pure CRDT functions from sub-project A.

**Tech Stack:** Web Crypto (`crypto.subtle`) shared by both webviews, a Rust `tiny_http` listener in the Tauri shell, Node's `http` in the Electron shell, `@capacitor-mlkit/barcode-scanning` on Android.

## Global Constraints

- **PREREQUISITES:** Sub-projects A and B complete. The merge functions and the Android app must both exist.
- **The crypto and the wire format live in `desktop/ui/sync-crypto.js` and are shared by both devices.** Two implementations of a protocol drift; one cannot.
- **`crypto.subtle` requires a secure context.** The Tauri and Electron webviews qualify; Android qualifies only because `capacitor.config.json` sets `androidScheme: https` (sub-project B, Task 2). Do not change that setting.
- **Never log a key, a plaintext payload, or a QR URL.** A key in a log file is a key on disk in cleartext.
- **The listener answers exactly one path**, `POST /v1/sync`. Everything else is 404 with no body. It is not a file server.
- **Bind to the LAN interface, never `0.0.0.0` on an unknown interface set**, and choose an ephemeral port.
- All new pure logic is unit-tested with the existing `node --test` harness; the transport is verified by hand against two real devices.
- Wire format version is `1`. A mismatched version is refused, not guessed at.

Create `desktop/ui/sync-crypto.js`:

```js
/*
 * Kinvt-study — the sync wire format.
 *
 * There is no server and no account, so there is no identity provider to lean
 * on. The trust comes from somewhere better: the key is generated fresh on the
 * desktop and handed to the phone THROUGH A CAMERA. It never travels over the
 * network at all, which is what makes plain HTTP an acceptable transport here
 * — the payload inside it is already end-to-end encrypted.
 *
 * AES-256-GCM is used because it authenticates as well as encrypts. A wrong
 * key, a flipped bit, or a rewritten field all fail the authentication tag,
 * so `open` either returns exactly what was sealed or throws. There is no
 * middle case to reason about.
 *
 * This file is loaded by BOTH devices. One implementation cannot disagree
 * with itself.
 */
(function (global) {
  'use strict';

  var WIRE_VERSION = 1;
  var MAX_SKEW_MS = 5 * 60 * 1000;
  var NONCE_BYTES = 12;              // 96 bits, the size GCM is defined for

  function toB64u(bytes) {
    var bin = '';
    for (var i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i]);
    return btoa(bin).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
  }

  function fromB64u(s) {
    var b64 = String(s).replace(/-/g, '+').replace(/_/g, '/');
    while (b64.length % 4) b64 += '=';
    var bin = atob(b64);
    var out = new Uint8Array(bin.length);
    for (var i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
    return out;
  }

  function newKey() {
    var k = new Uint8Array(32);
    global.crypto.getRandomValues(k);
    return Promise.resolve(k);
  }

  function importKey(bytes) {
    return global.crypto.subtle.importKey('raw', bytes, { name: 'AES-GCM' }, false, ['encrypt', 'decrypt']);
  }

  function utf8(s) {
    var out = [];
    for (var i = 0; i < s.length; i++) {
      var c = s.charCodeAt(i);
      if (c < 0x80) out.push(c);
      else if (c < 0x800) out.push(0xc0 | (c >> 6), 0x80 | (c & 63));
      else out.push(0xe0 | (c >> 12), 0x80 | ((c >> 6) & 63), 0x80 | (c & 63));
    }
    return new Uint8Array(out);
  }

  function fromUtf8(bytes) {
    var s = '', i = 0;
    while (i < bytes.length) {
      var c = bytes[i++];
      if (c < 0x80) s += String.fromCharCode(c);
      else if (c < 0xe0) s += String.fromCharCode(((c & 31) << 6) | (bytes[i++] & 63));
      else s += String.fromCharCode(((c & 15) << 12) | ((bytes[i++] & 63) << 6) | (bytes[i++] & 63));
    }
    return s;
  }

  // `ts` is a parameter rather than read from the clock so that staleness is
  // directly testable.
  function seal(keyBytes, obj, ts) {
    var nonce = new Uint8Array(NONCE_BYTES);
    global.crypto.getRandomValues(nonce);
    var body = utf8(JSON.stringify({ ts: ts || Date.now(), body: obj }));

    return importKey(keyBytes).then(function (key) {
      return global.crypto.subtle.encrypt({ name: 'AES-GCM', iv: nonce }, key, body);
    }).then(function (ct) {
      return { v: WIRE_VERSION, nonce: toB64u(nonce), ct: toB64u(new Uint8Array(ct)) };
    });
  }

  function open(keyBytes, env) {
    if (!env || env.v !== WIRE_VERSION) {
      return Promise.reject(new Error('unsupported wire version'));
    }
    return importKey(keyBytes).then(function (key) {
      return global.crypto.subtle.decrypt(
        { name: 'AES-GCM', iv: fromB64u(env.nonce) }, key, fromB64u(env.ct)
      );
    }).then(function (plain) {
      var msg = JSON.parse(fromUtf8(new Uint8Array(plain)));
      // A replayed message is authentic — it was sealed with the real key —
      // so only its age gives it away.
      if (Math.abs(Date.now() - (msg.ts || 0)) > MAX_SKEW_MS) {
        throw new Error('message is stale');
      }
      return msg.body;
    });
  }

  global.KinvtSyncCrypto = {
    WIRE_VERSION: WIRE_VERSION,
    MAX_SKEW_MS: MAX_SKEW_MS,
    newKey: newKey,
    seal: seal,
    open: open,
    toB64u: toB64u,
    fromB64u: fromB64u
  };
})(typeof window !== 'undefined' ? window : globalThis);
```

```js
/*
 * Kinvt-study — pairing, and remembering the devices you have paired with.
 *
 * The pairing URL is what the QR code contains. It carries the key, so it is
 * never logged, never put in a page title, and expires quickly: a bystander
 * who photographs the screen has a couple of minutes at most, and the listener
 * is closed by then anyway.
 *
 * Once paired, the key is stored so later syncs need no scan. That key is
 * therefore at rest in app storage, and is exactly as protected as the device
 * is — which is why "Forget device" exists.
 */
(function (global) {
  'use strict';

  var S = global.KinvtStorage;
  var C = global.KinvtSyncCrypto;
  var SCHEME = 'kinvt1://';
  var PAIRING_TTL_MS = 2 * 60 * 1000;
  var KEY_BYTES = 32;

  function buildUrl(o) {
    return SCHEME + o.host + ':' + o.port + '/?k=' + C.toB64u(o.key) +
      '&d=' + encodeURIComponent(o.deviceId) + '&e=' + o.expiresAt;
  }

  function parseUrl(url) {
    var s = String(url || '');
    if (s.indexOf(SCHEME) !== 0) throw new Error('not a Kinvt pairing code (bad scheme)');

    var rest = s.slice(SCHEME.length);
    var q = rest.indexOf('?');
    if (q === -1) throw new Error('pairing code has no key');

    var hostPort = rest.slice(0, q).replace(/\/$/, '');
    var colon = hostPort.lastIndexOf(':');
    if (colon === -1) throw new Error('pairing code has no port');

    var params = {};
    rest.slice(q + 1).split('&').forEach(function (kv) {
      var i = kv.indexOf('=');
      if (i > 0) params[kv.slice(0, i)] = decodeURIComponent(kv.slice(i + 1));
    });

    if (!params.k) throw new Error('pairing code has no key');
    if (!params.d) throw new Error('pairing code has no device id');

    var key = C.fromB64u(params.k);
    if (key.length !== KEY_BYTES) throw new Error('pairing key is the wrong length');

    var expiresAt = parseInt(params.e, 10) || 0;
    if (Date.now() > expiresAt) throw new Error('pairing code has expired — show a new one');

    return {
      host: hostPort.slice(0, colon),
      port: parseInt(hostPort.slice(colon + 1), 10),
      key: key,
      deviceId: params.d,
      expiresAt: expiresAt
    };
  }

  function allPeers() { return S.read(S.KEYS.peers, {}); }

  function savePeer(deviceId, keyBytes, name) {
    var peers = allPeers();
    peers[deviceId] = {
      key: C.toB64u(keyBytes),
      name: name || deviceId,
      lastSyncAt: (peers[deviceId] && peers[deviceId].lastSyncAt) || 0
    };
    S.write(S.KEYS.peers, peers);
  }

  function getPeer(deviceId) {
    var p = allPeers()[deviceId];
    if (!p) return null;
    return { key: C.fromB64u(p.key), name: p.name, lastSyncAt: p.lastSyncAt || 0 };
  }

  function listPeers() {
    var peers = allPeers();
    return Object.keys(peers).map(function (id) {
      return { deviceId: id, name: peers[id].name, lastSyncAt: peers[id].lastSyncAt || 0 };
    });
  }

  function forgetPeer(deviceId) {
    var peers = allPeers();
    delete peers[deviceId];
    S.write(S.KEYS.peers, peers);
  }

  function touchPeer(deviceId, at) {
    var peers = allPeers();
    if (!peers[deviceId]) return;
    peers[deviceId].lastSyncAt = at;
    S.write(S.KEYS.peers, peers);
  }

  global.KinvtPairing = {
    PAIRING_TTL_MS: PAIRING_TTL_MS,
    buildUrl: buildUrl,
    parseUrl: parseUrl,
    savePeer: savePeer,
    getPeer: getPeer,
    listPeers: listPeers,
    forgetPeer: forgetPeer,
    touchPeer: touchPeer
  };
})(typeof window !== 'undefined' ? window : globalThis);
```

```js
const MODULES = ['merge.js', 'storage.js', 'progress.js', 'selection.js',

test('two devices converge after one exchange', async () => {
  const desktop = device();

  desktop.KinvtProgress.recordAnswer(q('up-1', 'upsc'), true);
  desktop.KinvtProgress.recordAnswer(q('up-2', 'upsc'), false);
  phone.KinvtProgress.recordAnswer(q('ss-1', 'ssc'), true);

  const req = await phone.KinvtSyncCrypto.seal(key, {

  phone.KinvtSync.applyRemote(back.payload);

  const dt = desktop.KinvtMerge.totals(desktop.KinvtProgress.getStats());
  const pt = phone.KinvtMerge.totals(phone.KinvtProgress.getStats());
```

Create `desktop/ui/sync-session.js`:

```js
/*
 * Kinvt-study — one sync conversation.
 *
 * Deliberately transport-free. `handleRequest` takes an envelope and returns
 * an envelope; `syncWith` takes a function that moves an envelope and gets one
 * back. Sockets, Rust listeners and fetch all live outside this file.
 *
 * That is what lets the whole protocol — including two devices converging and
 * an attacker being rejected — be tested in one process with no network.
 */
(function (global) {
  'use strict';

  var C = global.KinvtSyncCrypto;
  var P = global.KinvtProgress;
  var Pair = global.KinvtPairing;

  var PAYLOAD_VERSION = 2;

  function localPayload() {
    return P.exportPayload();          // already {version: 2, stats, review, settings}
  }

  function applyRemote(payload) {
    return P.importPayload(payload);   // merges; idempotent by construction
  }

  // The listener's entire decision-making. `findKey(deviceId)` returns the
  // shared key for a known peer, or null.
  //
  // Failures are deliberately indistinguishable to the caller beyond a status
  // code: an unknown device and a wrong key both yield 401 with no body, so a
  // prober learns nothing about which device ids are paired.
  function handleRequest(envelope, findKey) {
    if (!envelope || envelope.v !== C.WIRE_VERSION || !envelope.nonce || !envelope.ct) {
      return Promise.resolve({ status: 400 });
    }

    // The device id is not inside the ciphertext, so it cannot be trusted —
    // it only selects which key to TRY. Authentication is the GCM tag.
    var claimed = envelope.d;
    var key = findKey(claimed);
    if (!key) return Promise.resolve({ status: 401 });

    return C.open(key, envelope).then(function (req) {
      if (!req || !req.payload || req.payload.version !== PAYLOAD_VERSION) {
        return { status: 422 };
      }
      var res = applyRemote(req.payload);
      if (!res.ok) return { status: 422 };

      return C.seal(key, {
        deviceId: global.KinvtStorage.deviceId('dsk'),
        payload: localPayload()
      }).then(function (out) {
        return { status: 200, envelope: out };
      });
    }).catch(function () {
      // A wrong key, tampering, or a stale timestamp all land here. No detail
      // is returned, because the difference is only useful to an attacker.
      return { status: 401 };
    });
  }

  function syncWith(peer, transport) {
    var url = 'http://' + peer.host + ':' + peer.port + '/v1/sync';

    return C.seal(peer.key, {
      deviceId: global.KinvtStorage.deviceId('and'),
      payload: localPayload()
    }).then(function (envelope) {
      envelope.d = global.KinvtStorage.deviceId('and');
      return transport(url, envelope);
    }).then(function (reply) {
      return C.open(peer.key, reply);
    }).then(function (msg) {
      if (!msg || !msg.payload) throw new Error('the other device sent nothing usable');
      var res = applyRemote(msg.payload);
      if (!res.ok) throw new Error(res.error);
      Pair.touchPeer(peer.deviceId, Date.now());
      return { ok: true };
    }).catch(function (e) {
      return { ok: false, error: 'could not reach the other device: ' + (e && e.message ? e.message : e) };
    });
  }

  global.KinvtSync = {
    PAYLOAD_VERSION: PAYLOAD_VERSION,
    localPayload: localPayload,
    applyRemote: applyRemote,
    handleRequest: handleRequest,
    syncWith: syncWith
  };
})(typeof window !== 'undefined' ? window : globalThis);
```

Note: `handleRequest` reads `envelope.d` for key selection, so `seal` results carry `d` — set by the caller, as `syncWith` does. Add the same in the test's manual seals if a test fails on a missing `d`; the tests above pass `findKey` as a constant so it does not matter there.

```rust
use std::sync::{Arc, Mutex};
use std::sync::mpsc;

/// The sync listener.
///
/// Rust owns the socket and nothing else: it does not decrypt, merge or even
/// parse the payload. The envelope is handed to the webview, which holds the
/// only implementation of the protocol, and the webview's answer is written
/// back. Putting the crypto here would mean writing it twice — once in Rust
/// for the desktop and once in JS for the phone — and two implementations of a
/// protocol drift.
struct SyncState {
    stop: Option<mpsc::Sender<()>>,
}

#[tauri::command]
fn sync_listen(app: tauri::AppHandle, state: tauri::State<'_, Arc<Mutex<SyncState>>>)
    -> Result<serde_json::Value, String>
{
    // Bind to port 0 and let the OS choose: a fixed port would be a
    // predictable target and could collide with something already running.
    let server = tiny_http::Server::http("0.0.0.0:0").map_err(|e| e.to_string())?;
    let port = server.server_addr().to_ip().ok_or("no ip")?.port();

    let (tx, rx) = mpsc::channel::<()>();
    state.lock().unwrap().stop = Some(tx);

    let handle = app.clone();
    std::thread::spawn(move || {
        for request in server.incoming_requests() {
            if rx.try_recv().is_ok() {
                break;
            }
            // One path only. This is not a file server, and anything else
            // gets a bare 404 with no hint that a real endpoint exists.
            if request.method() != &tiny_http::Method::Post || request.url() != "/v1/sync" {
                let _ = request.respond(tiny_http::Response::empty(404));
                continue;
            }
            let _ = handle.emit("sync-request", ());
            let _ = request.respond(tiny_http::Response::empty(503));
        }
    });

    Ok(serde_json::json!({ "host": local_ip(), "port": port }))
}

#[tauri::command]
fn sync_stop(state: tauri::State<'_, Arc<Mutex<SyncState>>>) {
    if let Some(tx) = state.lock().unwrap().stop.take() {
        let _ = tx.send(());
    }
}

/// The LAN address to put in the QR code. Loopback is useless to a phone.
fn local_ip() -> String {
    use std::net::UdpSocket;
    // No packet is actually sent; connecting a UDP socket just asks the OS
    // which interface it would use to reach the internet, which is the
    // interface the phone is also on.
    UdpSocket::bind("0.0.0.0:0")
        .and_then(|s| { s.connect("8.8.8.8:80")?; s.local_addr() })
        .map(|a| a.ip().to_string())
        .unwrap_or_else(|_| "127.0.0.1".to_string())
}
```

Register both commands in `invoke_handler`, and `.manage(Arc::new(Mutex::new(SyncState { stop: None })))` on the builder.

**Implementation note for the agent:** the skeleton above responds 503 to prove the socket works. Completing it means passing the request body to the webview and writing its reply back — do this by giving the thread a `mpsc` channel pair per request and having `sync_respond` send the JSON back through it. Do not move the crypto into Rust.

```js
const http = require('node:http');
let syncServer = null;

// Same contract as the Tauri commands, so ui/ cannot tell the shells apart.
ipcMain.handle('sync_listen', async () => {
  if (syncServer) syncServer.close();
  syncServer = http.createServer((req, res) => {
    if (req.method !== 'POST' || req.url !== '/v1/sync') {
      res.writeHead(404).end();
      return;
    }
    let body = '';
    // A sync payload is a few hundred kilobytes at most. Anything larger is
    // not a sync, and reading it would be an easy way to exhaust memory.
    req.on('data', (c) => {
      body += c;
      if (body.length > 4 * 1024 * 1024) { res.writeHead(413).end(); req.destroy(); }
    });
    req.on('end', async () => {
      try {
        const reply = await settingsWin.webContents.executeJavaScript(
          `window.KinvtSync.handleRequest(${JSON.stringify(JSON.parse(body))}, ` +
          `(d) => { const p = window.KinvtPairing.getPeer(d); return p && p.key; })`
        );
        res.writeHead(reply.status, { 'content-type': 'application/json' });
        res.end(reply.envelope ? JSON.stringify(reply.envelope) : '');
      } catch (e) {
        res.writeHead(400).end();
      }
    });
  });
  await new Promise((r) => syncServer.listen(0, '0.0.0.0', r));
  return { host: lanAddress(), port: syncServer.address().port };
});

ipcMain.handle('sync_stop', () => {
  if (syncServer) { syncServer.close(); syncServer = null; }
});

function lanAddress() {
  const nets = require('node:os').networkInterfaces();
  for (const name of Object.keys(nets)) {
    for (const n of nets[name]) {
      // Skip loopback and IPv6: the QR must contain an address the phone can
      // actually reach.
      if (n.family === 'IPv4' && !n.internal) return n.address;
    }
  }
  return '127.0.0.1';
}
```

```js
/*
 * Kinvt-study — pairing and syncing from the phone.
 *
 * The camera is the security boundary here: scanning the code is what proves
 * the two devices are in the same room, which is why the key can travel that
 * way and never over the network.
 */
(function (global) {
  'use strict';
  if (!global.Capacitor || !global.Capacitor.Plugins) return;

  var Scanner = global.Capacitor.Plugins.BarcodeScanner;

  function pair() {
    return Scanner.requestPermissions()
      .then(function (p) {
        if (p.camera !== 'granted') throw new Error('camera permission is needed to scan the code');
        return Scanner.scan({ formats: ['QR_CODE'] });
      })
      .then(function (result) {
        var raw = result.barcodes && result.barcodes[0] && result.barcodes[0].rawValue;
        if (!raw) throw new Error('no code found');
        var peer = global.KinvtPairing.parseUrl(raw);   // throws on expiry or tampering
        global.KinvtPairing.savePeer(peer.deviceId, peer.key, 'Desktop');
        return syncWith(peer);
      });
  }

  function transport(url, envelope) {
    return fetch(url, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify(envelope)
    }).then(function (r) {
      if (!r.ok) throw new Error('the other device refused the sync (' + r.status + ')');
      return r.json();
    });
  }

  function syncWith(peer) { return global.KinvtSync.syncWith(peer, transport); }

  // A repeat sync needs no scan, but it does need the desktop listening and
  // its address, which changes with the network — so the QR is still the way
  // back if the address moved.
  function syncSaved() {
    var peers = global.KinvtPairing.listPeers();
    if (!peers.length) return Promise.resolve({ ok: false, error: 'no paired device yet' });
    return Promise.resolve({ ok: false, error: 'scan the code on the desktop to sync' });
  }

  global.KinvtScan = { pair: pair, syncSaved: syncSaved };
})(typeof window !== 'undefined' ? window : globalThis);
```
